const digest = async (path) =>
  createHash("sha256")
    .update(await readFile(path))
    .digest("hex");
async function filesUnder(directory) {
  const out = [];
  for (const entry of await readdir(directory, { withFileTypes: true })) {
    const path = resolve(directory, entry.name);
    if (entry.isDirectory()) out.push(...(await filesUnder(path)));
    else out.push(path);
  }
  return out;
}

    "packages/storage/sqlite/dist/sqlite3.wasm",
    "node_modules/pdfjs-dist/build/pdf.worker.mjs",
    "packages/documents/tooling/pdfjs-range-patch.mjs",
  ];
  for (const dir of [
    "packages/documents/src",
    "packages/storage/src",
    "packages/core/src",
  ])
    for (const path of await filesUnder(resolve(root, dir)))
      if (/\.(ts|json)$/.test(path)) paths.push(relative(root, path));
  for (const n of [1, 100, 1000])
    paths.push(`packages/documents/tests/fixtures/pages-${n}.pdf`);
  return Object.fromEntries(
    await Promise.all(
      paths
        .sort()
        .map(async (path) => [path, await digest(resolve(root, path))]),
    ),
  );
}
function liveChildPids() {
  const rows = execFileSync("ps", ["-axo", "pid=,ppid="], { encoding: "utf8" })
    .trim()
    .split("\n")
    .map((line) => line.trim().split(/\s+/).map(Number));
  return rows
    .filter(([pid, parent]) => {
      if (parent !== process.pid) return false;
      try {
        process.kill(pid, 0);
        return true;
      } catch {
        return false;
      }
    })
    .map(([pid]) => pid);
}
function memorySampler(pid) {
  const data = {
    browserPid: pid,
    baselineRssKiB: null,
    peakRssKiB: null,
    peakProcessCount: 0,
    unavailable: null,
    samples: [],
    droppedSamples: 0,
  };
  const started = performance.now();
  const sample = () => {
    try {
      const rows = execFileSync("ps", ["-axo", "pid=,ppid=,rss="], {
        encoding: "utf8",
        timeout: 3000,
      })
        .trim()
        .split("\n")
        .map((line) => line.trim().split(/\s+/).map(Number));
      if (
        !rows.some((row) => row[0] === pid) ||
        rows.some((row) => row.length !== 3 || !row.every(Number.isFinite))
      )
        throw new Error("ps process-tree output unavailable");
      const ids = new Set([pid]);
      let changed = true;
      while (changed) {
        changed = false;
        for (const [child, parent] of rows)
          if (ids.has(parent) && !ids.has(child)) {
            ids.add(child);
            changed = true;
          }
      }
      const selected = rows.filter((row) => ids.has(row[0]));
      const rssKiB = selected.reduce((sum, row) => sum + row[2], 0);
      data.baselineRssKiB ??= rssKiB;
      data.peakRssKiB = Math.max(data.peakRssKiB ?? rssKiB, rssKiB);
      data.peakProcessCount = Math.max(data.peakProcessCount, selected.length);
      if (data.samples.length < 2400)
        data.samples.push({
          elapsedMs: performance.now() - started,
          rssKiB,
          processCount: selected.length,
        });
      else data.droppedSamples++;
    } catch (error) {
      data.unavailable ??= String(error);
    }
  };
  sample();
  const timer = setInterval(sample, 500);
  return {
    stop() {
      clearInterval(timer);
      sample();
      return data;
    },
  };
}

    logLevel: "warn",

  if (!address || typeof address === "string")
    throw new Error("No isolated origin");
  let origin = `http://127.0.0.1:${address.port}`;
  report.origin = origin;
  report.origins = [origin];
  await save();
  for (const [name, engine] of selected) {

    for (const key of report.method.order) {
      if (!server) {
        server = await preview({
          configFile: false,
          root: import.meta.dirname,
          logLevel: "warn",
          build: { outDir },
          preview: { host: "127.0.0.1", port: 0, strictPort: true },
        });
        const nextAddress = server.httpServer.address();
        if (!nextAddress || typeof nextAddress === "string")
          throw new Error("No isolated case origin");
        origin = `http://127.0.0.1:${nextAddress.port}`;
        if (report.origins.includes(origin))
          throw new Error("Case origin was reused");
        report.origins.push(origin);
      }
      const beforeChildren = new Set(liveChildPids());
      const profile = resolve(temporary, `profile-${name}-${key}`);
      const context = await engine.launchPersistentContext(
        profile,
        { headless: true, ...(name === "chromium" && key.includes("dense") ? {args:["--remote-debugging-port=0"]} : {}) },
      );
      const browserPids = liveChildPids().filter(
        (pid) => !beforeChildren.has(pid),
      );
      const browserPid = browserPids.length === 1 ? browserPids[0] : null;
      const page = await context.newPage();
      const record = {
        fixture: key,
        origin,
        status: "running",
        startedAt: new Date().toISOString(),
        loadAverage: loadavg(),
        externalRequests: [],
        pageErrors: [],
      };
      host.cases.push(record);
      await save();
      let memory, heap;
      context.on("request", (request) => {
        if (
          !request.url().startsWith(origin + "/") &&
          record.externalRequests.length < 20
        )
          record.externalRequests.push(request.url());
      });
      page.on("pageerror", (error) => {
        if (record.pageErrors.length < 20)
          record.pageErrors.push(String(error));
      });
      try {
        await page.goto(origin);
        await page.waitForFunction(() => !!window.documentPerf);
        record.userAgent = await page.evaluate(() => navigator.userAgent);
        record.initialOpfs = await page.evaluate(async () => {
          const names = [];
          for await (const entry of (
            await navigator.storage.getDirectory()
          ).values())
            names.push(entry.name);
          return names;
        });
        if (record.initialOpfs.length)
          throw new Error("Case must start with fresh OPFS");
        const setupStart = performance.now();
        record.source = await page.evaluate(
          (key) => window.documentPerf.setup(key),
          key.replace(/^cancel-/, ""),
        );
        record.setupMs = performance.now() - setupStart;
        if(key.includes("dense")) { const fixture=report.denseFixtures[`${key.replace(/^cancel-/, "")}.pdf`]; if(record.source.sourceSha256!==fixture.sha256 || record.source.sourceBytes!==fixture.bytes) throw new Error("Dense fixture hash/length mismatch"); }
        memory = memorySampler(browserPid);
        if (key.includes("dense")) {
          if(name === "chromium") { try { heap = await startHeapSampler(profile,origin); } catch(error) { record.heap={available:false,reason:String(error)}; } }
          else record.heap={available:false,reason:"WebKit does not expose Chromium CDP Runtime.getHeapUsage; descendant RSS excludes OS-managed XPC services on macOS."};
        }
        console.log(
          `${name} ${key}: production persistence measurement started`,
        );
        let timeout;
        try {
          record.extraction = await Promise.race([
            page.evaluate(
              (cancel) => window.documentPerf.run(cancel),
              key.startsWith("cancel-"),
            ),
            new Promise((_, reject) => {
              timeout = setTimeout(
                () =>
                  reject(new Error("Bounded 20-minute case deadline exceeded")),
                1200000,
              );
            }),
          ]);
        } finally {
          clearTimeout(timeout);
        }
        if(heap) { record.heap=await heap.stop(); heap=undefined; }
        record.memory = memory.stop();
        memory = undefined;
        if (record.externalRequests.length || record.pageErrors.length)
          throw new Error("Unexpected browser errors or external requests");
        const expectedPages =
          key === "image" ? 1 : Number(key.replace(/^cancel-/, "").replace(/^dense-/, ""));
        if (record.extraction.pageCount !== expectedPages)
          throw new Error("Fixture page count mismatch");
        if(key.includes("dense") && !record.extraction.failure && (record.extraction.maxPageUTF16<100000 || record.extraction.maxPageUTF16>200000 || record.extraction.totalTextUTF16 < expectedPages*100000)) throw new Error("Dense output workload was not attained");
        record.status =
          !key.startsWith("cancel-") && record.extraction.failure
            ? "rejected"
            : "passed";
        console.log(
          `${name} ${key} ${record.status}: ${record.extraction.totalExtractionMs.toFixed(1)} ms total, first indexed ${record.extraction.firstIndexedPageMs?.toFixed(1)} ms, RSS peak ${record.memory.peakRssKiB} KiB`,
        );
        await page.evaluate(() => window.documentPerf.close());
      } catch (error) {
        record.status = "failed";
        record.error = String(error?.stack ?? error);
        throw error;
      } finally {
        if(heap) record.heap=await heap.stop();
        if (memory) record.memory = memory.stop();
        record.finishedAt = new Date().toISOString();
        await context.close();
        await new Promise((resolve) => server.httpServer.close(resolve));
        server = undefined;
        await save();
      }
    }
    host.status = host.cases.some((record) => record.status === "rejected")
      ? "measured_with_rejections"
      : "passed";
    await save();
  }
  const after = await fingerprint();
  report.sourceStable =
    JSON.stringify(after) === JSON.stringify(report.sourceSha256);
  if (!report.sourceStable) {
    report.sourceSha256After = after;
    throw new Error("Measured source changed during capture");
  }
  report.status = report.hosts.some(
    (host) => host.status === "measured_with_rejections",
  )
    ? "measured_with_rejections"
    : "passed";
} catch (error) {
  report.status = "failed";
  report.error = String(error?.stack ?? error);
  process.exitCode = 1;
  console.error(report.error);
} finally {
  report.finishedAt = new Date().toISOString();
  report.endLoadAverage = loadavg();
  await save();
  if (server) await new Promise((resolve) => server.httpServer.close(resolve));
  await rm(temporary, { recursive: true, force: true });
}
